  };
  export type UnifonicSendResult = {
    ok: boolean;
    provider: "unifonic" | "twilio-fallback";
    unifonicMessageId?: string;             // From Unifonic's MessageID field on success
    error?: string;
    errorCode?: "MISSING_CREDS" | "UNIFONIC_4xx" | "UNIFONIC_5xx" | "NETWORK_ERROR" | "PARSE_ERROR";
  };
  export async function sendSmsUnifonic(args: UnifonicSendArgs): Promise<UnifonicSendResult>;
  // Implementation rules:
  //  - POST https://api.unifonic.com/rest/SMS/messages
  //  - Content-Type: application/x-www-form-urlencoded
  //  - Body: AppSid=<appSid>&SenderID=<senderId>&Recipient=<phone>&Body=<URL-encoded message>
  //  - On 200 with Unifonic JSON: { success: "true", data: { MessageID: "..." } } → return ok:true with unifonicMessageId
  //  - On 200 with success="false": return ok:false with errorCode="UNIFONIC_4xx" + Unifonic error.message
  //  - On 4xx HTTP: return ok:false with errorCode="UNIFONIC_4xx"
  //  - On 5xx: errorCode="UNIFONIC_5xx"
  //  - On fetch throw: errorCode="NETWORK_ERROR"
  //  - Phone normalisation: phone.startsWith("+") ? phone.slice(1) : phone (Unifonic accepts without +)
  //  - UTF-8 Arabic: encodeURIComponent(message) — verify Khaleeji round-trips intact

  //  - If UNIFONIC_APP_SID env var absent → caller uses existing notificationChannels.sendSms (Twilio).

backend/src/services/outboundCostMeter.ts MUST export:
  export type OutboundChannelForCost = "IN_APP_PUSH" | "WHATSAPP" | "SMS_UNIFONIC" | "SMS_TWILIO_INTL";
  export const KD_PER_CHANNEL: Record<OutboundChannelForCost, number> = {
    IN_APP_PUSH: 0,           // Expo free tier
    WHATSAPP: 0.0010,         // $0.0034 utility template in Kuwait at 0.305 KWD/USD
    SMS_UNIFONIC: 0.012,      // Local route, per Unifonic Kuwait pricing
    SMS_TWILIO_INTL: 0.097,   // $0.3164 × 0.305
  };
  export async function getDailySpentKd(tenantId: string): Promise<number>;
  export async function tryReserve(tenantId: string, channel: OutboundChannelForCost, estimatedKd?: number): Promise<{ allowed: boolean; spentKd: number; limitKd: number | null }>;
  // Implementation rules:
  //  - getDailySpentKd: prisma.notificationDelivery.findMany where tenantId, sentAt within today's Kuwait-local window (UTC+3, midnight reset). Reduce: sum estimated KD by channel+provider lookup.
  //  - tryReserve: read Tenant.outboundDailyKdLimit (null = unlimited → always allowed). Compute next-spend = currentSpent + (estimatedKd ?? KD_PER_CHANNEL[channel]). Return allowed = nextSpend <= limit.
  //  - Kuwait-local midnight: `(new Date().getUTCHours() + 3) % 24` for hour-of-day; start of day = today's UTC midnight - 3h.
  //  - NO race-protection in v1 — we accept that concurrent sends in the same second might both pass the check. Cost overage in worst case = 1 message worth, acceptable. Wave 4 may add Redis SETNX-based reservation.

      export type UnifonicSendResult = {

        unifonicMessageId?: string;
        error?: string;
        errorCode?: UnifonicErrorCode;

      const UNIFONIC_API_URL = "https://api.unifonic.com/rest/SMS/messages";

      export async function sendSmsUnifonic(args: UnifonicSendArgs): Promise<UnifonicSendResult> {
        if (!args.appSid || args.appSid.trim() === "") {
          return { ok: false, provider: "unifonic", errorCode: "MISSING_CREDS", error: "appSid required" };
        }
        if (!args.senderId || args.senderId.trim() === "") {
          return { ok: false, provider: "unifonic", errorCode: "MISSING_CREDS", error: "senderId required" };
        }

        const phone = args.phone.startsWith("+") ? args.phone.slice(1) : args.phone;
        const params = new URLSearchParams();
        params.set("AppSid", args.appSid);
        params.set("SenderID", args.senderId);
        params.set("Recipient", phone);
        params.set("Body", args.message);  // URLSearchParams handles UTF-8 encoding for Arabic

        let res: Response;
        try {
          res = await fetch(UNIFONIC_API_URL, {
            method: "POST",
            headers: { "Content-Type": "application/x-www-form-urlencoded" },
            body: params.toString(),
          });
        } catch (e: any) {
          logger.warn({ err: e?.message }, "unifonic network error");
          return { ok: false, provider: "unifonic", errorCode: "NETWORK_ERROR", error: e?.message ?? String(e) };
        }

        if (res.status >= 500) {
          return { ok: false, provider: "unifonic", errorCode: "UNIFONIC_5xx", error: `HTTP ${res.status}` };
        }

        const json = await res.json().catch(() => null);
        if (!json) {
          return { ok: false, provider: "unifonic", errorCode: "PARSE_ERROR", error: "bad unifonic response" };
        }

        // Unifonic returns { success: "true" | "false", data: { MessageID, ... } | { Code, Message } }
        if (!res.ok || json.success === "false") {
          const errMsg = json?.errorCode ? `${json.errorCode}: ${json.message}` : (json?.message ?? `HTTP ${res.status}`);
          return { ok: false, provider: "unifonic", errorCode: "UNIFONIC_4xx", error: errMsg };
        }

        const messageId = json?.data?.MessageID;
        return { ok: true, provider: "unifonic", unifonicMessageId: messageId };
      }

      export type OutboundChannelForCost = "IN_APP_PUSH" | "WHATSAPP" | "SMS_UNIFONIC" | "SMS_TWILIO_INTL";

      export const KD_PER_CHANNEL: Record<OutboundChannelForCost, number> = {
        IN_APP_PUSH: 0,
        WHATSAPP: 0.0010,         // ~$0.0034 utility/Kuwait at 0.305 KWD/USD
        SMS_UNIFONIC: 0.012,      // Unifonic local Kuwait
        SMS_TWILIO_INTL: 0.097,   // Twilio international $0.3164

      const KUWAIT_OFFSET_HRS = 3;

      function kuwaitStartOfDay(now: Date = new Date()): Date {
        // Kuwait midnight in UTC = today's Kuwait-local midnight - 3h
        const utcMs = now.getTime();
        const kwOffsetMs = KUWAIT_OFFSET_HRS * 60 * 60 * 1000;
        const kwNow = new Date(utcMs + kwOffsetMs);
        kwNow.setUTCHours(0, 0, 0, 0);
        return new Date(kwNow.getTime() - kwOffsetMs);
      }

      function channelToCostKey(channel: string, provider: string | null): OutboundChannelForCost {
        if (channel === "WHATSAPP") return "WHATSAPP";
        if (channel === "SMS") return provider === "twilio" ? "SMS_TWILIO_INTL" : "SMS_UNIFONIC";
        return "IN_APP_PUSH";
      }

      export async function getDailySpentKd(tenantId: string): Promise<number> {
        const since = kuwaitStartOfDay();
        const rows = await prisma.notificationDelivery.findMany({
          where: { tenantId, sentAt: { gte: since }, status: "SENT" },
          select: { channel: true, provider: true },
        });
        return rows.reduce((sum, r) => sum + (KD_PER_CHANNEL[channelToCostKey(r.channel, r.provider)] ?? 0), 0);
      }

      export async function tryReserve(
        tenantId: string,
        channel: OutboundChannelForCost,
        estimatedKd?: number,
      ): Promise<{ allowed: boolean; spentKd: number; limitKd: number | null }> {
        const tenant = await prisma.tenant.findFirst({
          where: { id: tenantId },
          select: { outboundDailyKdLimit: true },
        });
        const limit = tenant?.outboundDailyKdLimit?.toNumber() ?? null;
        if (limit === null) return { allowed: true, spentKd: 0, limitKd: null };

        const spent = await getDailySpentKd(tenantId);
        const cost = estimatedKd ?? KD_PER_CHANNEL[channel];
        const allowed = spent + cost <= limit;
        return { allowed, spentKd: spent, limitKd: limit };
      }

      const KUWAIT_OFFSET_HRS = 3;

      function parseHHmm(s: string): number | null {
        const m = s.match(/^(\d{2}):(\d{2})$/);
        if (!m) return null;
        const h = Number(m[1]);
        const mm = Number(m[2]);
        if (h < 0 || h > 23 || mm < 0 || mm > 59) return null;
        return h * 60 + mm;
      }

      export function isQuietHours(args: {
        tenant: { quietHoursStart: string | null; quietHoursEnd: string | null };
        now?: Date;
        severity?: "LOW" | "MEDIUM" | "HIGH" | "CRITICAL";
      }): boolean {
        if (args.severity === "CRITICAL") return false;
        const { quietHoursStart, quietHoursEnd } = args.tenant;
        if (!quietHoursStart || !quietHoursEnd) return false;

        const start = parseHHmm(quietHoursStart);
        const end = parseHHmm(quietHoursEnd);
        if (start === null || end === null) return false;
        if (start === end) return false;  // empty window

        const now = args.now ?? new Date();
        const kwMinutes = ((now.getUTCHours() + KUWAIT_OFFSET_HRS) * 60 + now.getUTCMinutes()) % (24 * 60);

        if (start > end) {
          // Wrap-around (e.g., 22:00..07:00 — quiet spans midnight)
          return kwMinutes >= start || kwMinutes < end;
        }
        // Normal window (e.g., 13:00..15:00 afternoon nap)
        return kwMinutes >= start && kwMinutes < end;
      }

          async (job: Job<SmsOptOutInboundJobData>) => {

                sentAt: new Date(receivedAt),
                sourceType: "INBOUND_OPT_OUT",
              },
            });

            return { ok: true, matched: true, driversUpdated: updated.count };
          },
