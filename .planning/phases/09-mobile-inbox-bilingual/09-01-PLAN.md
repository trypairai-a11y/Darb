      import { logger } from "../config/logger";

      const EXPO_PUSH_URL = "https://exp.host/--/api/v2/push/send";

      export type ExpoPushMessage = {
        to: string;
        title: string;
        body: string;
        data?: Record<string, unknown>;
        badge?: number;
        channelId?: string;
        sound?: "default" | null;
      };

      export type ExpoPushReceipt = {
        status: "ok" | "error";
        id?: string;
        message?: string;
        details?: { error?: string };
      };

      async function postWithRetry(payload: ExpoPushMessage[], attempt = 0): Promise<ExpoPushReceipt[]> {
        const headers: Record<string, string> = {
          "Accept": "application/json",
          "Accept-Encoding": "gzip, deflate",
          "Content-Type": "application/json",
        };
        if (process.env.EXPO_ACCESS_TOKEN) {
          headers["Authorization"] = `Bearer ${process.env.EXPO_ACCESS_TOKEN}`;
        }

        let res: Response;
        try {
          res = await fetch(EXPO_PUSH_URL, { method: "POST", headers, body: JSON.stringify(payload) });
        } catch (e: any) {
          logger.warn({ err: e?.message }, "expo-push network error");
          return payload.map(() => ({ status: "error", message: "network error" }));
        }

        if (res.status === 429 && attempt < 3) {
          const wait = 1000 * Math.pow(2, attempt);
          await new Promise((r) => setTimeout(r, wait));
          return postWithRetry(payload, attempt + 1);
        }

        if (!res.ok) {
          const text = await res.text().catch(() => "");
          logger.warn({ status: res.status, text }, "expo-push non-ok response");
          return payload.map(() => ({ status: "error", message: `HTTP ${res.status}` }));
        }

        const json = await res.json().catch(() => null);
        if (!json || !Array.isArray(json.data)) {
          return payload.map(() => ({ status: "error", message: "bad expo response" }));
        }
        return json.data;
      }

      export async function sendExpoPush(messages: ExpoPushMessage[]): Promise<ExpoPushReceipt[]> {
        if (messages.length === 0) return [];
        // Default channelId for Android (Pitfall 1).
        const normalised = messages.map((m) => ({
          channelId: m.channelId ?? "darb-inbox",
          sound: m.sound === null ? null : (m.sound ?? "default"),
          ...m,
        }));
        // Expo accepts batches up to 100.
        const out: ExpoPushReceipt[] = [];
        for (let i = 0; i < normalised.length; i += 100) {
          const batch = normalised.slice(i, i + 100);
          const receipts = await postWithRetry(batch);
          out.push(...receipts);
        }
        return out;
      }

      const KUWAIT_OFFSET_HRS = 3;

      function isQuietHours(now: Date = new Date()): boolean {
        const kwHour = (now.getUTCHours() + KUWAIT_OFFSET_HRS) % 24;
        return kwHour >= 22 || kwHour < 6;
      }

      export async function resolveOutboundChannels(args: {
        driverId: string;
        tenantId: string;
        severity: Severity;
      }): Promise<ResolvedOutbound | null> {
        const driver = await prisma.driver.findFirst({
          where: { id: args.driverId, tenantId: args.tenantId },
          select: {
            preferredLanguage: true,
            outboundOptIn: true,
            smsOptOut: true,
            expoPushToken: true,
            phone: true,
            tenant: { select: {
              defaultLanguage: true,
              outboundChannels: true,
              whatsappPhoneNumberId: true,
            } },
          },
        });
        if (!driver) return null;
        if (!driver.outboundOptIn) return null;

        const lang: "EN" | "AR" =
          driver.preferredLanguage === "EN" ? "EN" :
          driver.preferredLanguage === "AR" ? "AR" :
          (driver.tenant.defaultLanguage as "EN" | "AR");

        const flags = (driver.tenant.outboundChannels ?? {}) as { inApp?: boolean; whatsapp?: boolean; sms?: boolean };
        const quiet = isQuietHours();
        const isCritical = args.severity === "CRITICAL";

        const channels: Channel[] = [];
        if (flags.inApp !== false && driver.expoPushToken) channels.push("IN_APP_PUSH");
        if (flags.whatsapp && driver.tenant.whatsappPhoneNumberId && driver.phone) channels.push("WHATSAPP");
        if (flags.sms && driver.phone && !driver.smsOptOut && (!quiet || isCritical)) channels.push("SMS");

        return {
          lang,
          body: "",  // resolveOutboundChannels does not pick the body; deliver() does
          channels,
          quietHoursSuppressed: quiet && !isCritical,
        };
      }

        tenantId: string;
        notificationId: string;
        driverId: string;
        bodyEn: string;
        bodyAr: string;
        channelHint?: "WHATSAPP" | "SMS" | "IN_APP" | "AUTO";
        severity: Severity;
        intent?: string;
      }): Promise<{ enqueued: Channel[]; suppressed: boolean }> {
        const resolved = await resolveOutboundChannels({
          driverId: args.driverId,
          tenantId: args.tenantId,
          severity: args.severity,
        });
        if (!resolved) return { enqueued: [], suppressed: true };

        const body = resolved.lang === "AR" ? args.bodyAr : args.bodyEn;

        // Stage-1 default: MEDIUM only enqueues push. HIGH/CRITICAL fans out all available.
        const fanOut = args.severity === "HIGH" || args.severity === "CRITICAL";
        const targets: Channel[] = fanOut ? resolved.channels : resolved.channels.filter((c) => c === "IN_APP_PUSH");

        const enqueued: Channel[] = [];

        // 1. IN_APP_PUSH — direct, synchronous, fire-and-forget (no BullMQ).
        if (targets.includes("IN_APP_PUSH")) {
          const driver = await prisma.driver.findFirst({
            where: { id: args.driverId, tenantId: args.tenantId },
            select: { expoPushToken: true, name: true },
          });
          if (driver?.expoPushToken) {
            const receipts = await sendExpoPush([{
              to: driver.expoPushToken,
              title: resolved.lang === "AR" ? "العمليات" /* العمليات */ : "Operations",
              body,
              channelId: "darb-inbox",
              data: { notificationId: args.notificationId, intent: args.intent ?? null },
            });
            const r = receipts[0];
            if (r?.status === "error" && r?.details?.error === "DeviceNotRegistered") {
              await prisma.driver.update({ where: { id: args.driverId }, data: { expoPushToken: null } });
              logger.info({ driverId: args.driverId }, "expo token nulled — DeviceNotRegistered");
            }
            if (r?.status === "ok") enqueued.push("IN_APP_PUSH");
          }
        }

        // 2. WHATSAPP / 3. SMS — stubbed in Wave 1 (logged only); Waves 2 + 3 wire the real adapters
        //    through notificationService.enqueueNotification + notificationChannels.sendWhatsAppCloud /
        //    sendSmsUnifonic. Wave 1 keeps these branches present so callers don't have to change

        if (targets.includes("WHATSAPP")) {
          logger.info({ tenantId: args.tenantId, notificationId: args.notificationId, intent: args.intent }, "outbound:whatsapp queued (stub — Wave 2)");
          // No DB write yet — Wave 2 will create NotificationDelivery row + call sendWhatsAppCloud.
        }
        if (targets.includes("SMS")) {
          logger.info({ tenantId: args.tenantId, notificationId: args.notificationId, intent: args.intent }, "outbound:sms queued (stub — Wave 3)");
        }

        return { enqueued, suppressed: false };
      }

      export interface InboxNotification {
        id: string;
        title: string;
        titleAr: string | null;
        message: string;
        bodyAr: string | null;
        type: string;
        category: string | null;

        createdAt: string;
      }

      const DB_NAME = "darb-outbox.db";
      const MAX_ROWS = 50;
      let _db: SQLite.SQLiteDatabase | null = null;

      async function db(): Promise<SQLite.SQLiteDatabase> {
        if (_db) return _db;
        _db = await SQLite.openDatabaseAsync(DB_NAME);
        await _db.execAsync(`
          CREATE TABLE IF NOT EXISTS inbox_cache (
            id TEXT PRIMARY KEY,
            payload TEXT NOT NULL,
            createdAt INTEGER NOT NULL,
            isRead INTEGER NOT NULL DEFAULT 0
          );
          CREATE INDEX IF NOT EXISTS idx_inbox_cache_createdAt ON inbox_cache(createdAt DESC);
        `);
        return _db;
      }

      export async function setCachedInbox(items: InboxNotification[]): Promise<void> {
        const conn = await db();
        const trimmed = items
          .slice()
          .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
          .slice(0, MAX_ROWS);
        await conn.withTransactionAsync(async () => {
          await conn.execAsync(`DELETE FROM inbox_cache;`);
          for (const item of trimmed) {
            await conn.runAsync(
              `INSERT INTO inbox_cache (id, payload, createdAt, isRead) VALUES (?, ?, ?, ?);`,
              [item.id, JSON.stringify(item), new Date(item.createdAt).getTime(), item.read ? 1 : 0]
            );
          }
        });
      }

      export async function getCachedInbox(): Promise<InboxNotification[]> {
        try {

      export async function registerPushToken(): Promise<{ token: string | null; changed: boolean }> {
        const perm = await Notifications.getPermissionsAsync();
        if (perm.status !== "granted") {
          const req = await Notifications.requestPermissionsAsync();
          if (req.status !== "granted") return { token: null, changed: false };
        }

        const projectId =
          (Constants?.expoConfig?.extra as any)?.eas?.projectId ??
          (Constants?.easConfig as any)?.projectId;

          throw new Error("Expo projectId missing — set extra.eas.projectId in app.json");
        }

        let tokenResp;
        try {
          tokenResp = await Notifications.getExpoPushTokenAsync({ projectId });
        } catch (e: any) {
          // Clear stored token on Expo error so next attempt re-registers cleanly (Pitfall 4 edge).
          await SecureStore.deleteItemAsync(LAST_TOKEN_KEY).catch(() => undefined);
          return { token: null, changed: false };
        }
        const token = tokenResp.data;
        const last = await SecureStore.getItemAsync(LAST_TOKEN_KEY);
        if (token === last) return { token, changed: false };

        await apiRegisterPushToken(token);
        await SecureStore.setItemAsync(LAST_TOKEN_KEY, token);
        return { token, changed: true };
      }
