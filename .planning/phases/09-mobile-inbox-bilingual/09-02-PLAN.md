<objective>
Wave 2 — WhatsApp Cloud API direct integration + per-tenant template registry + founder onboarding scripts.
Stage-2 of Phase 9: turn the Wave 1 WHATSAPP stub branch into a real Meta-direct send path. Ship the
12 utility templates (6 intents × EN + Khaleeji AR), the per-tenant WABA setup script, and the template
registration tooling so the founder can onboard a tenant's WhatsApp Business Account in ~10 minutes
once Meta Business Verification has cleared.

Purpose: Wave 1 left WHATSAPP as a `logger.info("stub — Wave 2")` line in `outboundResolver.deliver`.
This Wave fills it in. Why direct Meta Cloud API (NOT Twilio-as-BSP): per RESEARCH §"WhatsApp BSP
choice" Meta-direct costs ~$0.0034/utility-template in Kuwait vs Twilio's $0.0084/msg (BSP markup
+ template fee) — a 60% reduction at fleet scale. Meta-direct also gives full template control
(submit + approve via Meta Business Manager, no Twilio intermediary). The existing Twilio
`sendWhatsApp` path stays as a tenant-by-tenant fallback for tenants without a Meta WABA (handled
by the `notificationWorker` branch logic).

Critical scope reminder: Phase 9 sends EVERY agent-initiated message as an approved template
(Pitfall 5 — Meta's 24h session window blocks free-form agent-originated sends). The 6 templates
in this Wave cover every agent action that produces a courier Notification today:
- `LATE_CLOCKIN_WARN` — Phase 2 attendance violation flow
- `GPS_STALE_WARN` — Phase 5/Phase 4 GPS-stale alerts (also Wave 1 of Phase 9)
- `ORDER_REJECTION_WARN` — Phase 4 violation engine
- `PENALTY_APPLIED` — Phase 4/Phase 7 penalty flow
- `SCHEDULE_CHANGED` — Phase 3 shift management
- `CASH_REMINDER` — Phase 6 cash reconciliation flow

Each template has a UTILITY-category Meta registration (cheapest tier, ~24h approval per Meta).
Marketing-category templates are explicitly OUT OF SCOPE for Phase 9 (PRD non-goal).

WABA ownership decision: Darb owns one WABA per tenant (resolved AUTONOMOUSLY in source-of-truth
context per the orchestrator). The Tenant.whatsappBusinessAccountId column points at the WABA Darb
provisioned for that tenant; Tenant.whatsappPhoneNumberId points at the Meta-issued phone number
registered to that WABA. The founder runs `wabaTenantSetup.ts` once after Meta Business Verification
clears, then `wabaTemplateRegister.ts` once to push the 12 template definitions for approval.

  export type WhatsappCloudSendArgs = {
    phone: string;                       // E.164 or local; gets leading '+' stripped before send
    templateName: string;                // e.g., "darb_late_clockin_warn"
    templateLang: "en" | "ar";           // ISO 639-1 lowercase
    variables: string[];                 // positional body parameters (Meta requires text-type only)
    phoneNumberId: string;               // Tenant.whatsappPhoneNumberId
    accessToken: string;                 // Tenant.settings.whatsapp.accessToken (decrypted at call site)
  };
  export type WhatsappCloudResult = {
    ok: boolean;
    provider: "whatsapp-cloud";
    metaMessageId?: string;              // wamid.HBg... — for audit trace
    error?: string;
    errorCode?: string;                  // "FREE_FORM_BLOCKED" | "META_4xx" | "NETWORK_ERROR" | "MISSING_CREDS"
  };
  export async function sendWhatsAppCloud(args: WhatsappCloudSendArgs): Promise<WhatsappCloudResult>;
  // Implementation rules:
  //  - POST https://graph.facebook.com/v22.0/{phoneNumberId}/messages
  //  - Headers: { "Content-Type": "application/json", "Authorization": `Bearer ${accessToken}` }
  //  - Body: { messaging_product: "whatsapp", to: <phone with '+' stripped>, type: "template",
  //            template: { name, language: { code }, components: [{ type: "body", parameters: variables.map(v => ({type:"text", text:v})) }] } }
  //  - ALWAYS template; if templateName missing or empty string → return {ok:false, errorCode:"FREE_FORM_BLOCKED"} WITHOUT calling fetch (Pitfall 5)
  //  - On 200 → parse messages[0].id → return {ok:true, metaMessageId}
  //  - On 4xx → parse error.message → return {ok:false, errorCode:"META_4xx", error}
  //  - On fetch throw → return {ok:false, errorCode:"NETWORK_ERROR", error}
  //  - Phone normalisation: phone.replace(/^\+/, "") — Meta wants no leading +
  //  - UTF-8: JSON.stringify naturally encodes Arabic; verify Khaleeji renders correctly across the wire (contract test asserts no \uXXXX escapes leak)

      export type WhatsappCloudErrorCode =
        | "FREE_FORM_BLOCKED"
        | "MISSING_CREDS"
        | "META_4xx"
        | "META_5xx"

      export type WhatsappCloudResult = {

        errorCode?: WhatsappCloudErrorCode;
      };

      const META_API_VERSION = "v22.0";

      export async function sendWhatsAppCloud(args: WhatsappCloudSendArgs): Promise<WhatsappCloudResult> {
        // Pitfall 5 — Meta's 24h session window blocks free-form sends. Phase 9 ALWAYS sends templates.
        if (!args.templateName || args.templateName.trim() === "") {
          return { ok: false, provider: "whatsapp-cloud", errorCode: "FREE_FORM_BLOCKED", error: "templateName required (free-form blocked)" };
        }
        if (!args.phoneNumberId || !args.accessToken) {
          return { ok: false, provider: "whatsapp-cloud", errorCode: "MISSING_CREDS", error: "phoneNumberId and accessToken required" };
        }

        const url = `https://graph.facebook.com/${META_API_VERSION}/${args.phoneNumberId}/messages`;
        const phone = args.phone.replace(/^\+/, ""); // Meta wants no leading +
        const body = {
          messaging_product: "whatsapp",
          to: phone,
          type: "template",
          template: {
            name: args.templateName,
            language: { code: args.templateLang },
            components: args.variables.length
              ? [
                  {
                    type: "body",
                    parameters: args.variables.map((v) => ({ type: "text", text: v })),
                  },
                ]
              : [],
          },
        };

        let res: Response;
        try {
          res = await fetch(url, {
            method: "POST",
            headers: {
              "Content-Type": "application/json",
              "Authorization": `Bearer ${args.accessToken}`,
            },
            body: JSON.stringify(body),
          });
        } catch (e: any) {
          logger.warn({ err: e?.message, template: args.templateName }, "whatsapp-cloud network error");
          return { ok: false, provider: "whatsapp-cloud", errorCode: "NETWORK_ERROR", error: e?.message ?? String(e) };
        }

        if (res.status >= 500) {
          return { ok: false, provider: "whatsapp-cloud", errorCode: "META_5xx", error: `HTTP ${res.status}` };
        }
        if (!res.ok) {
          const errJson = await res.json().catch(() => ({}));
          const errMsg = errJson?.error?.message ?? `HTTP ${res.status}`;
          return { ok: false, provider: "whatsapp-cloud", errorCode: "META_4xx", error: errMsg };
        }

        const json = await res.json().catch(() => null);
        if (!json || !Array.isArray(json.messages) || !json.messages[0]?.id) {
          return { ok: false, provider: "whatsapp-cloud", errorCode: "PARSE_ERROR", error: "bad meta response shape" };
        }
        return { ok: true, provider: "whatsapp-cloud", metaMessageId: json.messages[0].id };
      }
