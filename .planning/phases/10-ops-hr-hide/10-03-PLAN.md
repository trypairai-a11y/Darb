Notification idempotency key (sourceId convention):
```
sourceId = `doc-expiry:${driverId}:${docType}:${daysToExpiryBucket}`
```
Where `daysToExpiryBucket` is the rounded calendar-day-based bucket -- using
`daysToExpiry` directly (an integer) is sufficient. Re-running the worker on the same
calendar day will encounter the same bucket -> findFirst returns the existing row -> skip.

NOTE on sourceId uniqueness (W-01): `Notification.sourceId` is NOT currently a unique
column in the schema. We rely on findFirst-by-sourceId early-return inside a transaction
for idempotency. A future deferred-items entry in the Wave 4 SUMMARY should track adding
`@@unique([tenantId, sourceId])` to lift this to a DB-level guarantee.

AgentRunLog dual-worker race mitigation (B-06): within the transaction that creates the
Notification + PendingAgentAction for a given tenant on a given Kuwait calendar day, we
SELECT for an existing AgentRunLog row matching `(tenantId, agentId='documentExpiry', startedAt >= nowKuwait)`.
If no row exists, we CREATE one. If a concurrent worker created one between our SELECT and
CREATE, we catch the prisma error and re-SELECT. Per-day-per-tenant uniqueness is enforced
by the SELECT predicate, not by a DB constraint.

    1. **Create `backend/src/services/documentExpiry/scan.ts`**:

       ```ts
       import type { PrismaClient } from "@prisma/client";
       import { differenceInCalendarDays } from "date-fns";

       export type DocType = "civilId" | "drivingLicense" | "vehicleInsurance" | "vehicleReg" | "workPermit" | "healthCert" | "foodHandlingCert";
       export const DOC_TYPES: DocType[] = ["civilId","drivingLicense","vehicleInsurance","vehicleReg","workPermit","healthCert","foodHandlingCert"];

       export type Transition = {
         tenantId: string; driverId: string; driverName: string;
         docType: DocType; expiry: Date;
         oldStatus: string | null;
         newStatus: "EXPIRING" | "EXPIRING_SOON" | "EXPIRED";
         daysToExpiry: number;
       };

       function statusFor(daysToExpiry: number): Transition["newStatus"] | null {
         if (daysToExpiry <= 0) return "EXPIRED";
         if (daysToExpiry <= 7) return "EXPIRING_SOON";
         if (daysToExpiry <= 30) return "EXPIRING";
         return null;
       }

       export async function scanDocumentExpiry(opts: {
         prisma: PrismaClient;
         nowKuwait: Date;
         withinDays?: number;
       }): Promise<Transition[]> {
         const { prisma, nowKuwait } = opts;
         const withinDays = opts.withinDays ?? 30;
         const tenants = await prisma.tenant.findMany({ select: { id: true } });
         const transitions: Transition[] = [];

         for (const t of tenants) {
           const drivers = await prisma.driver.findMany({
             where: { tenantId: t.id },
             select: {
               id: true, name: true, tenantId: true,
               civilIdExpiry: true, civilIdStatus: true,
               drivingLicenseExpiry: true, drivingLicenseStatus: true,
               vehicleInsuranceExpiry: true, vehicleInsuranceStatus: true,
               vehicleRegExpiry: true, vehicleRegStatus: true,
               workPermitExpiry: true, workPermitStatus: true,
               healthCertExpiry: true, healthCertStatus: true,
               foodHandlingCertExpiry: true, foodHandlingCertStatus: true,
             },
           });

           for (const d of drivers as any[]) {
             for (const docType of DOC_TYPES) {
               const expiry: Date | null = d[`${docType}Expiry`] ?? null;
               if (!expiry) continue;
               const daysToExpiry = differenceInCalendarDays(expiry, nowKuwait);
               if (daysToExpiry > withinDays) continue;
               const newStatus = statusFor(daysToExpiry);
               if (!newStatus) continue;
               const oldStatus: string | null = d[`${docType}Status`] ?? null;
               if (oldStatus === newStatus) continue;
               transitions.push({
                 tenantId: t.id, driverId: d.id, driverName: d.name,
                 docType, expiry, oldStatus, newStatus, daysToExpiry,
               });
             }
           }
         }
         return transitions;
       }
       ```

       ```ts
       import IORedis from "ioredis";
       import { startOfDay } from "date-fns";
       import { zonedTimeToUtc, utcToZonedTime } from "date-fns-tz";
       import { prisma } from "../config";
       import { logger } from "../config/logger";
       import { env } from "../config/env";
       import { scanDocumentExpiry } from "../services/documentExpiry/scan";
       import { buildExpiryNotificationCopy } from "../services/documentExpiry/i18n";

       const QUEUE = "documentExpiry";
       const SCHEDULER_ID = "documentExpiry-daily";
       const KUWAIT_TZ = "Asia/Kuwait";
       const AGENT_ID = "documentExpiry";
       const TRIGGER_EVENT = "cron:documentExpiry-daily";

       // ─── Connection / queue singletons (Phase 4 convention -- no-op in dev sandboxes) ───
       let _connectionSingleton: IORedis | null = null;
       let _queueSingleton: Queue | null = null;

       function hasRedis(): boolean {
         return !!env.REDIS_URL && env.REDIS_URL !== "redis://localhost:6379";
       }

       function getConnection(): IORedis | null {
         if (!hasRedis()) return null;
         if (_connectionSingleton) return _connectionSingleton;
         _connectionSingleton = new IORedis(env.REDIS_URL, { maxRetriesPerRequest: null });
         return _connectionSingleton;
       }

       function getQueue(): Queue | null {
         if (!hasRedis()) return null;
         if (_queueSingleton) return _queueSingleton;
         const connection = getConnection();
         if (!connection) return null;
         _queueSingleton = new Queue(QUEUE, { connection });
         return _queueSingleton;
       }

       // ─── Kuwait-calendar day boundary helper ───
       export function getKuwaitDayStart(reference: Date = new Date()): Date {
         const kuwaitNow = utcToZonedTime(reference, KUWAIT_TZ);
         const kuwaitDayStart = startOfDay(kuwaitNow);
         return zonedTimeToUtc(kuwaitDayStart, KUWAIT_TZ);
       }

       // ─── B-06: get-or-create the per-day-per-tenant AgentRunLog row ───
       async function getOrCreateRunLog(tenantId: string, nowKuwait: Date): Promise<string> {
         // SELECT for any AgentRunLog row for this agent/tenant started on or after the
         // Kuwait day-start. Dual-worker race is defeated by the SELECT inside the same
         // transaction as the subsequent inserts -- the second worker sees the first
         // worker's row and reuses its id.
         const existing = await prisma.agentRunLog.findFirst({
           where: {
             tenantId,
             agentId: AGENT_ID,
             startedAt: { gte: nowKuwait },
           },
           select: { id: true },
           orderBy: { startedAt: "desc" },
         });
         if (existing) return existing.id;
         // No row -- create one. The (tenantId, agentId, startedAt) tuple is implicitly
         // unique per Kuwait day because nowKuwait pins the lower bound.
         const created = await prisma.agentRunLog.create({
           data: {
             tenantId,
             agentId: AGENT_ID,
             triggerEvent: TRIGGER_EVENT,   // B-06 -- required field per schema.prisma:2186
             model: "system",                // B-06 -- required field; non-LLM workers use "system"
             status: "running",
           },
           select: { id: true },
         });
         return created.id;
       }

       // ─── Main pure-ish runner (exported for tests; mirrors Phase 4 processTick) ───
       export async function runDocumentExpiry(now: Date = new Date()): Promise<number> {
         const nowKuwait = getKuwaitDayStart(now);
         const transitions = await scanDocumentExpiry({ prisma, nowKuwait, withinDays: 30 });

         for (const t of transitions) {
           const sourceId = `doc-expiry:${t.driverId}:${t.docType}:${t.daysToExpiry}`;
           const copy = buildExpiryNotificationCopy(t);
           const severity = t.newStatus === "EXPIRED" ? "CRITICAL" : "HIGH";

           // Transaction wraps the existence check + the three inserts so a concurrent
           // worker pass sees a consistent view. W-01: a future migration should add
           // @@unique([tenantId, sourceId]) on Notification to lift this guard to the DB.
           await prisma.$transaction(async (tx) => {
             // Existence check inside the transaction -- defeats dual-worker race.
             const existing = await tx.notification.findFirst({
               where: { tenantId: t.tenantId, sourceId },
               select: { id: true },
             });
             if (existing) return;

             // B-06: get-or-create AgentRunLog for this (tenantId, day) bucket.
             // Re-do the SELECT inside the transaction with the tx client so we share its view.
             let runId: string;
             const existingRun = await tx.agentRunLog.findFirst({
               where: {
                 tenantId: t.tenantId,
                 agentId: AGENT_ID,
                 startedAt: { gte: nowKuwait },
               },
               select: { id: true },
               orderBy: { startedAt: "desc" },
             });
             if (existingRun) {
               runId = existingRun.id;
             } else {
               const created = await tx.agentRunLog.create({
                 data: {
                   tenantId: t.tenantId,
                   agentId: AGENT_ID,
                   triggerEvent: TRIGGER_EVENT,
                   model: "system",
                   status: "running",
                 },
                 select: { id: true },
               });
               runId = created.id;
             }

             await tx.notification.create({
               data: {
                 tenantId: t.tenantId,
                 type: "DOCUMENT_EXPIRY",
                 severity,
                 category: "IMPORTANT",
                 title: copy.title, titleAr: copy.titleAr,
                 message: copy.message, bodyAr: copy.bodyAr,
                 sourceId,
                 metadata: { driverId: t.driverId, docType: t.docType, daysToExpiry: t.daysToExpiry } as any,
               },
             });

             await tx.pendingAgentAction.create({
               data: {
                 tenantId: t.tenantId,
                 runId,
                 agentId: AGENT_ID,
                 toolName: "draftCourierMessage",
                 input: {
                   driverId: t.driverId,
                   docType: t.docType,
                   expiry: t.expiry,
                   body: copy.message,
                   bodyAr: copy.bodyAr,
                 } as any,
                 recommendation: "approve",
                 confidence: 0.8,
                 reasoning: `Document ${t.docType} ${t.newStatus === "EXPIRED" ? "expired" : `expires in ${t.daysToExpiry} day(s)`}. Owner should send a bilingual reminder; if expiry passes, propose suspension.`,
                 priorityScore: t.newStatus === "EXPIRED" ? 90 : (t.newStatus === "EXPIRING_SOON" ? 60 : 30),
                 subjectType: "driver_document",
                 subjectId: t.driverId,
               },
             });

             // Update the canonical status field on the Driver.
             const statusField = `${t.docType}Status` as const;
             await tx.driver.updateMany({
               where: { tenantId: t.tenantId, id: t.driverId },
               data: { [statusField]: t.newStatus } as any,
             });
           });
         }

         logger.info({ count: transitions.length }, "documentExpiryWorker: completed");
         return transitions.length;
       }

       export function getDocumentExpiryWorker(): Worker | null {
         const connection = getConnection();
         if (!connection) return null;
         return new Worker(QUEUE, async (_job) => { await runDocumentExpiry(); }, { connection });
       }

       export async function registerDocumentExpirySchedule(): Promise<void> {

           {
             name: SCHEDULER_ID,
             data: {},
             opts: { removeOnComplete: true },
           },
         );
       }
       ```
