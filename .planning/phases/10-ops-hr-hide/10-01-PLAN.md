       ```ts
       import { Request, Response, NextFunction } from "express";
       import { prisma } from "../config";
       import { logger } from "../config/logger";

       export type PlatformKey = "KEETA" | "TALABAT" | "DELIVEROO" | "AMERICANA";

       export function requirePlatformEnabled(platform: PlatformKey) {
         return async (req: Request, res: Response, next: NextFunction): Promise<void> => {
           if (!req.user?.tenantId) {
             res.status(401).json({ error: "Tenant context required" });
             return;
           }
           // ADMIN role + super-admin always bypass (Pitfall §3).
           if (req.user.role === "ADMIN" || (req.user as any).isSuperAdmin === true) {
             next();
             return;
           }
           try {
             const tenant = await (prisma as any).tenant?.findUnique?.({
               where: { id: req.user.tenantId },
               select: { enabledPlatforms: true },
             });
             const enabled: string[] = tenant?.enabledPlatforms ?? [
               "KEETA", "TALABAT", "DELIVEROO", "AMERICANA",
             ];
             if (!enabled.includes(platform)) {
               // 404, NO body -- Pitfall §1.
               res.status(404).end();
               return;
             }
             next();
           } catch (err) {
             logger.warn({ err, tenantId: req.user.tenantId }, "requirePlatformEnabled: DB lookup failed, denying request");
             res.status(404).end();
           }
         };
       }
       ```

       ```tsx
       "use client";
       import { useState, useEffect } from "react";
       import { useRole } from "@/hooks/useRole";
       import { useAuth } from "@/contexts/AuthContext";
       import { useEnabledPlatforms, type PlatformKey } from "@/hooks/useEnabledPlatforms";
       import { notFound } from "next/navigation";
       import api from "@/lib/api";

       const PLATFORMS: PlatformKey[] = ["KEETA","TALABAT","DELIVEROO","AMERICANA"];

       export default function TenantSettingsPage() {
         const { isAdmin } = useRole();
         const { user } = useAuth();
         const { raw } = useEnabledPlatforms();
         const [draft, setDraft] = useState<PlatformKey[]>(raw);

                 <div className="w-full max-w-md rounded bg-white p-6 shadow">
                   <h3 className="text-lg font-semibold">Disable all platforms?</h3>
                   <p className="mt-2 text-sm">
                     Disabling all platforms hides them from non-admin users. ADMIN users (you) will still see them.
       ```
