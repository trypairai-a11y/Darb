DriverLeave model (new -- small):
```prisma
model DriverLeave {
  id        String   @id @default(uuid())
  tenantId  String
  driverId  String
  startDate DateTime
  endDate   DateTime
  reason    String?
  status    String   @default("PENDING")   // PENDING | APPROVED | REJECTED | CANCELLED
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  tenant    Tenant  @relation(fields: [tenantId], references: [id])
  driver    Driver  @relation(fields: [driverId], references: [id])
  @@index([tenantId, driverId])
  @@index([tenantId, startDate])
}
```

Note: DriverLeave is intentionally separate from the existing `LeaveRequest` model. Per
DEC-driver-vs-employee-split, drivers and office staff (Users) are different records. The
existing `LeaveRequest` continues to serve office staff per REQ-hr-leave; `DriverLeave`
serves driver leave events surfaced on the per-driver timeline and the `/hr/leave` tab.

NOTE on B-04 resolution (lint:tenant scope): DriverInventory.tenantId was added in Wave 0
specifically so the equipment route can write `where: { tenantId, ... }` at the top level.
The lint:tenant ESLint rule only inspects the top-level `where` clause -- relation filters
like `where: { driver: { tenantId } }` do NOT satisfy the rule. The equipment route in
Task 2.1 below uses the top-level pattern exclusively.

       import { Router } from "express";
       import { prisma } from "../../config";
       const router = Router();

       router.get("/:driverId/timeline", async (req, res) => {
         try {
           const tenantId = req.user!.tenantId;
           const { driverId } = req.params;

           // Confirm driver belongs to tenant -- defensive check.
           const owned = await prisma.driver.findFirst({
             where: { tenantId, id: driverId }, select: { id: true },
           });
           if (!owned) { res.status(404).json({ error: "Driver not found" }); return; }

           const [penalties, violations, restrictions, appeals] = await Promise.all([
             prisma.penalty.findMany({ where: { tenantId, driverId } }),
             prisma.violation.findMany({ where: { tenantId, driverId } }),
             prisma.driverRestriction.findMany({ where: { tenantId, driverId } }),
             prisma.appeal.findMany({ where: { tenantId, violation: { driverId } } }),
           ]);

           const events = [
             ...penalties.map((p: any)    => ({ kind: "PENALTY" as const,     at: p.createdAt,     data: p })),
             ...violations.map((v: any)   => ({ kind: "VIOLATION" as const,   at: v.violationTime, data: v })),
             ...restrictions.map((r: any) => ({ kind: "RESTRICTION" as const, at: r.startDate,     data: r })),
             ...appeals.map((a: any)      => ({ kind: "APPEAL" as const,      at: a.appealedAt,    data: a })),
           ].sort((a, b) => new Date(b.at).getTime() - new Date(a.at).getTime());

           res.json({ driverId, events });
         } catch (err: any) {
           res.status(500).json({ error: err.message });
         }
       });
       export default router;
       ```

       import { Router } from "express";
       import { z } from "zod";
       import { prisma } from "../../config";
       const router = Router();

       const ItemTypeEnum = z.enum(["HELMET","BIG_BAG","MOBILE_PHONE","SIM_CARD","PETROL_CARD","UNIFORM","OTHER"]);
       const ConditionEnum = z.enum(["NEW","GOOD","WORN","DAMAGED"]).optional();
       const createSchema = z.object({
         driverId: z.string(),
         itemType: ItemTypeEnum,
         serialNumber: z.string().nullable().optional(),
         condition: ConditionEnum,
         notes: z.string().optional(),
         issuedDate: z.coerce.date().optional(),
       });
       const updateSchema = z.object({
         condition: ConditionEnum,
         notes: z.string().nullable().optional(),
         returnedDate: z.coerce.date().nullable().optional(),
       });

       router.get("/", async (req, res) => {
         const tenantId = req.user!.tenantId;
         const driverId = req.query.driverId as string | undefined;
         // B-04: top-level tenantId filter so lint:tenant passes.
         const where: any = { tenantId };
         if (driverId) where.driverId = driverId;
         const rows = await prisma.driverInventory.findMany({
           where,
           orderBy: { createdAt: "desc" },
         });
         res.json({ rows });
       });

       router.post("/", async (req, res) => {
         try {
           const body = createSchema.parse(req.body);
           const tenantId = req.user!.tenantId;
           // Confirm the driver lives in the caller's tenant before stamping tenantId.
           const driver = await prisma.driver.findFirst({
             where: { tenantId, id: body.driverId },
             select: { id: true },
           });
           if (!driver) { res.status(404).json({ error: "Driver not found in tenant" }); return; }
           const row = await prisma.driverInventory.create({
             data: { ...body, tenantId, issued: true },
           });
           res.status(201).json(row);
         } catch (err: any) {
           if (err?.code === "P2002") { res.status(409).json({ error: "Duplicate (driverId, itemType, serialNumber)" }); return; }
           if (err?.issues) { res.status(400).json({ error: err.issues }); return; }
           res.status(500).json({ error: err.message });
         }
       });

       router.put("/:id", async (req, res) => {
         try {
           const tenantId = req.user!.tenantId;
           // B-04: top-level tenantId in the existence check.
           const existing = await prisma.driverInventory.findFirst({
             where: { tenantId, id: req.params.id },
             select: { id: true },
           });
           if (!existing) { res.status(404).json({ error: "Equipment row not found" }); return; }
           const body = updateSchema.parse(req.body);
           // updateMany scoped by tenantId so the update verb is also lint:tenant-clean.
           const result = await prisma.driverInventory.updateMany({
             where: { tenantId, id: req.params.id },
             data: body,
           });
           if (result.count === 0) { res.status(404).json({ error: "Equipment row not found" }); return; }
           const updated = await prisma.driverInventory.findFirst({
             where: { tenantId, id: req.params.id },
           });
           res.json(updated);
         } catch (err: any) {
           if (err?.issues) { res.status(400).json({ error: err.issues }); return; }
           res.status(500).json({ error: err.message });
         }
       });

       export default router;
       ```

    4. **Create `backend/src/routes/hr/documents.ts`**:

       ```ts
       import { Router } from "express";
       import { prisma } from "../../config";
       import { addDays } from "date-fns";
       const router = Router();

       const DOC_FIELDS = [
         "civilId","drivingLicense","vehicleInsurance","vehicleReg",
         "workPermit","healthCert","foodHandlingCert",
       ] as const;

       router.get("/expiring", async (req, res) => {
         const tenantId = req.user!.tenantId;
         const within = Math.min(Math.max(parseInt(String(req.query.within ?? "30"), 10) || 30, 1), 180);
         const today = new Date();
         const cutoff = addDays(today, within);

         const drivers = await prisma.driver.findMany({
           where: {
             tenantId,
             OR: DOC_FIELDS.map(field => ({
               [`${field}Expiry`]: { gte: today, lte: cutoff },
             })),
           },
           select: {
             id: true, name: true,
             civilIdExpiry: true, civilIdStatus: true,
             drivingLicenseExpiry: true, drivingLicenseStatus: true,
             vehicleInsuranceExpiry: true, vehicleInsuranceStatus: true,
             vehicleRegExpiry: true, vehicleRegStatus: true,
             workPermitExpiry: true, workPermitStatus: true,
             healthCertExpiry: true, healthCertStatus: true,
             foodHandlingCertExpiry: true, foodHandlingCertStatus: true,
           },
         });

         const rows = drivers.flatMap((d: any) => DOC_FIELDS.flatMap(field => {
           const exp = d[`${field}Expiry`];
           if (!exp) return [];
           const days = Math.ceil((new Date(exp).getTime() - today.getTime()) / (1000*60*60*24));
           if (days < 0 || days > within) return [];
           return [{
             driverId: d.id, driverName: d.name,
             documentType: field,
             expiry: exp,
             status: d[`${field}Status`] ?? "VALID",
             daysToExpiry: days,
           }];
         }));

         res.json({ drivers: rows });
       });

       export default router;
       ```

       Note: this endpoint uses naive `new Date()` for the cutoff -- same as
       `aiInsightsEngine.ts:680`. Wave 3 fixes timezone for the WORKER. Read-side request
       latency is acceptable with UTC cutoffs in Wave 2; the worker is what triggers
       calendar-sensitive Notification rows.
