From RESEARCH §"Pattern 3" — Holt-Winters reference implementation (copy verbatim into holtWinters.ts):
```typescript
export function holtWintersTriple(series: number[], period: number, alpha=0.3, beta=0.05, gamma=0.2, horizon=168) {
  const n = series.length;
  if (n < 2 * period) throw new Error(`Need ≥ 2 periods (${2 * period} points) for HW`);
  let level = series.slice(0, period).reduce((a, b) => a + b, 0) / period;
  const period2Avg = series.slice(period, 2 * period).reduce((a, b) => a + b, 0) / period;
  let trend = (period2Avg - level) / period;
  const seasonals = series.slice(0, period).map(v => (level ? v / level : 1));
  const fitted: number[] = [];
  for (let t = 0; t < n; t++) {
    const s = seasonals[t % period];
    fitted.push((level + trend) * s);
    if (t >= period) {
      const value = series[t];
      const prevLevel = level;
      level = alpha * (value / s) + (1 - alpha) * (level + trend);
      trend = beta * (level - prevLevel) + (1 - beta) * trend;
      seasonals[t % period] = gamma * (value / level) + (1 - gamma) * s;
    }
  }
  const forecast: number[] = [];
  for (let h = 1; h <= horizon; h++) forecast.push((level + h * trend) * seasonals[(n + h - 1) % period]);
  const residuals = fitted.map((f, i) => series[i] - f).slice(period);
  const meanRes = residuals.reduce((a, b) => a + b, 0) / residuals.length;
  const sd = Math.sqrt(residuals.reduce((a, b) => a + (b - meanRes) ** 2, 0) / residuals.length);
  const z = 1.645;
  return { fitted, forecast, ci90: { lower: forecast.map(f => Math.max(0, f - z * sd)), upper: forecast.map(f => f + z * sd) } };
}
```

          ```sql
          WHERE "tenantId" = ${tenantId}
            AND timestamp >= NOW() - INTERVAL '8 weeks'
            -- if zone provided: AND ("metadata"->>'zone') = ${opts.zone}
          GROUP BY 1
          ORDER BY 1 ASC
          ```
       2. Build `series: number[]` of length = hours_observed. Pad missing hours with 0.

       6. Project: `horizonN = sum_{d=1..N} (slope * (90 + d) + intercept)`.
