```typescript
import type { AgentAction } from "@prisma/client";
import { prisma } from "../../config";

export type Reverser = (
  audit: AgentAction,
  tenantId: string,
  userId: string,
) => Promise<void>;

// ─── Per-tool rollback windows (founder-revisitable per RESEARCH.md Open Q #5) ─
const WINDOW_DAYS: Record<string, number> = {
  applyPenalty: 30,
  suspendDriver: 7,
  draftCourierMessage: 5 / (60 * 24),     // 5 minutes = 5 / (60*24) of a day
  sendCourierMessage: 5 / (60 * 24),
  escalateToHumanSupervisor: 1,
  reassignShift: -1,                       // sentinel — check shift.scheduledStart
  createTrainingTask: -1,                  // sentinel — check trainingTask.status
  recordCashSettlement: 0,                 // sentinel — ALWAYS forbidden
  generatePayrollAdjustment: -1,           // sentinel — check payrollRun.status
};

export function toolHasReverser(toolName: string): boolean {
  return toolName in REVERSERS;
}

export async function auditWindowOk(
  audit: AgentAction,
  toolName: string,
): Promise<boolean> {
  const days = WINDOW_DAYS[toolName];
  if (days === undefined) return false;
  if (days === 0) return false;            // recordCashSettlement — never
  if (days > 0) {
    const ageMs = Date.now() - audit.createdAt.getTime();
    return ageMs <= days * 86_400_000;
  }
  // Sentinel cases — check entity state instead of clock
  if (toolName === "reassignShift") {
    const shiftId = (audit.originalProposal as any)?.shiftId
      ?? (audit.subjectType === "Shift" ? audit.subjectId : null);
    if (!shiftId) return false;
    const shift = await prisma.shift.findFirst({
      where: { id: shiftId, tenantId: audit.tenantId },
      select: { scheduledStart: true },
    });
    return shift ? shift.scheduledStart.getTime() > Date.now() : false;
  }
  if (toolName === "createTrainingTask") {
    const taskId = audit.subjectId;
    if (!taskId) return false;
    const task = await prisma.trainingTask.findFirst({
      where: { id: taskId, tenantId: audit.tenantId },
      select: { status: true },
    });
    return task ? task.status !== "COMPLETED" : false;
  }
  if (toolName === "generatePayrollAdjustment") {
    const adjustmentId = audit.subjectId;
    if (!adjustmentId) return false;
    const adj = await prisma.payrollAdjustment.findFirst({
      where: { id: adjustmentId, tenantId: audit.tenantId },
      select: { run: { select: { status: true } } },
    });
    return adj ? adj.run.status !== "FROZEN" : false;
  }
  return false;
}

// ─── REVERSERS table — per-tool soft rollback handlers ────────────────────────
//
// CON-audit-row-shape: NEVER prisma.<entity>.delete; always status flip.
// Tenant-scoped on every updateMany via {id, tenantId}.
//
// Phase 2 ships draftCourierMessage's reverser inline in audit.ts:163-249. Wave 4
// transcribes it here so audit.ts can dispatch via a single REVERSERS lookup.

export const REVERSERS: Record<string, Reverser> = {
  draftCourierMessage: async (audit, tenantId) => {
    if (!audit.agentRunId) return;
    await prisma.notification.updateMany({
      where: {
        tenantId,
        metadata: { path: ["drafterRunId"], equals: audit.agentRunId },
      },
      data: { type: "AGENT_DRAFT_ROLLED_BACK" },
    });
  },

  applyPenalty: async (audit, tenantId) => {
    const penaltyId =
      (audit.originalProposal as any)?.penaltyId
      ?? (audit.subjectType === "Penalty" ? audit.subjectId : null);
    if (!penaltyId) throw new Error("applyPenalty rollback missing penaltyId");
    await prisma.penalty.updateMany({
      where: { id: penaltyId, tenantId },
      data: { penaltyStatus: "OVERTURNED" },
    });
  },

  suspendDriver: async (audit, tenantId) => {
    if (audit.subjectType !== "Driver" || !audit.subjectId) {
      throw new Error("suspendDriver rollback missing driver subjectId");
    }
    await prisma.driver.updateMany({
      where: { id: audit.subjectId, tenantId },
      data: { status: "ACTIVE" },
    });
    // Also cancel the active DriverRestriction the tool created; the suspension cascade
    // included a TEMPORARY restriction with endDate = now + N days. Mark it ended now.
    await prisma.driverRestriction.updateMany({
      where: {
        tenantId,
        driverId: audit.subjectId,
        type: "TEMPORARY",
        endDate: { gt: new Date() },
      },
      data: { endDate: new Date() },
    });
    // Note: Shift.updateMany(BOOKED -> CANCELLED) from the original cascade is NOT
    // un-cancelled — re-booking a cancelled shift is the dispatcher's manual call.
  },

  reassignShift: async (audit, tenantId) => {
    const op = audit.originalProposal as any;
    const shiftId = op?.shiftId ?? (audit.subjectType === "Shift" ? audit.subjectId : null);
    const previousDriverId = op?.previousDriverId;
    if (!shiftId || !previousDriverId) {
      throw new Error("reassignShift rollback missing shiftId or previousDriverId");
    }
    await prisma.shift.updateMany({
      where: { id: shiftId, tenantId },
      data: { driverId: previousDriverId },
    });
  },

  recordCashSettlement: async () => {
    throw new Error(
      "Cash settlement rollback requires accountant manual reversal",
    );
  },

  sendCourierMessage: async (audit, tenantId) => {
    // B1/W1 — NotificationDeliveryStatus enum has no CANCELLED value (existing enum:
    // QUEUED|SENDING|SENT|FAILED|DEAD — schema.prisma:1714-1726). Soft rollback
    // strategy: flip the parent Notification.type prefix so Phase 9 worker skips
    // pending deliveries. If NotificationDelivery has already transitioned to SENT,
    // the message is already out — rollback is documentation-only at that point.
    if (audit.agentRunId) {
      await prisma.notification.updateMany({
        where: {
          tenantId,
          metadata: { path: ["drafterRunId"], equals: audit.agentRunId },
        },
        data: { type: "AGENT_SEND_ROLLED_BACK" },
      });
    }
    // No NotificationDelivery mutation — Phase 9 worker checks the parent
    // Notification.type before sending; if prefix matches AGENT_SEND_ROLLED_BACK,
    // the delivery row is left QUEUED but the worker drops it as a no-op.
  },

  createTrainingTask: async (audit, tenantId) => {
    const taskId =
      (audit.originalProposal as any)?.trainingTaskId
      ?? (audit.subjectType === "TrainingTask" ? audit.subjectId : null);
    if (!taskId) throw new Error("createTrainingTask rollback missing taskId");
    await prisma.trainingTask.updateMany({
      where: { id: taskId, tenantId },
      data: { status: "CANCELLED" },
    });
  },

  escalateToHumanSupervisor: async () => {
    // Information-only — no entity to undo. The audit.rolledBackAt stamp documents
    // the operator's intent to "withdraw" the escalation, but the recipients have
    // already been notified. No-op reverser.
    return;
  },

  generatePayrollAdjustment: async (audit, tenantId) => {
    const adjustmentId =
      (audit.originalProposal as any)?.adjustmentId
      ?? (audit.subjectType === "PayrollAdjustment" ? audit.subjectId : null);
    if (!adjustmentId) {
      throw new Error("generatePayrollAdjustment rollback missing adjustmentId");
    }
    const adj = await prisma.payrollAdjustment.findFirst({
      where: { id: adjustmentId, tenantId },
      select: { id: true, run: { select: { status: true } } },
    });
    if (!adj) throw new Error("PayrollAdjustment not in tenant scope");
    if (adj.run.status === "FROZEN") {
      throw new Error(
        "Payroll rollback forbidden — run is FROZEN; reverse via accountant counter-adjustment",
      );
    }
    await prisma.payrollAdjustment.updateMany({
      where: { id: adjustmentId, tenantId },
      data: { status: "REVERSED" },
    });
  },
};
```

```typescript
"use client";
import { useState } from "react";
import { ConfirmModal } from "@/components/shared/ConfirmModal";
import { rollbackAgentAction } from "@/lib/financeApi";
import type { AgentAction } from "@/types/decisions";

export function RollbackConfirmModal({
  audit,
  open,
  onClose,
  onSuccess,
}: {
  audit: AgentAction;
  open: boolean;
  onClose: () => void;
  onSuccess: () => void;
}) {
  const [reason, setReason] = useState("");
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handleConfirm() {
    if (reason.trim().length < 20) {
      setError("Reason must be at least 20 characters.");
      return;
    }
    setSubmitting(true);
    setError(null);
    try {
      await rollbackAgentAction(audit.id, reason.trim());
      onSuccess();
      onClose();
    } catch (err: any) {
      setError(err?.response?.data?.error ?? "Rollback failed");
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <ConfirmModal
      open={open}
      onClose={onClose}
      title={`Roll back ${audit.toolName}`}
      onConfirm={handleConfirm}
      confirmLabel={submitting ? "Rolling back…" : "Confirm rollback"}
      confirmDisabled={submitting}
    >
      <p className="text-sm text-sand-700 mb-2">
        This will soft-reverse the action. The audit row stays; the side effect is undone.
      </p>
      <label className="text-sm font-medium text-sand-900 block mb-1">Reason (min 20 chars)</label>
      <textarea
        className="w-full border border-sand-300 rounded-md px-2 py-1.5 text-sm"
        rows={3}
        value={reason}
        onChange={(e) => setReason(e.target.value)}
        placeholder="Why are you rolling this back?"
      />
      {error && <div className="mt-2 text-sm text-rose-600">{error}</div>}
    </ConfirmModal>
  );
}
```
