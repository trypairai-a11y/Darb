**Step A — Create `backend/src/agent/tools/action/applySuspension.ts` (W4 resolution — single source of truth helper):**

```typescript
// REQ-agent-action-tools (Phase 8 Wave 1)
// Shared cascade helper: Driver.status -> SUSPENDED + Shift.updateMany(future BOOKED -> CANCELLED)
// + DriverRestriction.create. Consumed by both applyPenalty (ACCOUNT_SUSPENSION) and suspendDriver.
// Single source of truth — any future cascade additions land HERE, not in two places.

import type { Prisma } from "@prisma/client";

export interface ApplySuspensionArgs {
  tenantId: string;
  driverId: string;
  reason: string;
  durationDays: number;  // capped 1..365 by callers
}

export async function applySuspension(
  tx: Prisma.TransactionClient,
  args: ApplySuspensionArgs,
): Promise<{ restrictionId: string; cancelledShiftCount: number }> {
  const { tenantId, driverId, reason, durationDays } = args;

  // 1. Flip Driver status (tenant-scoped via where clause)
  await tx.driver.updateMany({
    where: { id: driverId, tenantId },
    data: { status: "SUSPENDED" },
  });

  // 2. Cancel future BOOKED shifts only (past shifts preserved for payroll)
  const cancelResult = await tx.shift.updateMany({
    where: {
      tenantId,
      driverId,
      scheduledStart: { gte: new Date() },
      status: "BOOKED",
    },
    data: { status: "CANCELLED" },
  });

  // 3. Create temporary restriction
  const startDate = new Date();
  const endDate = new Date(startDate.getTime() + durationDays * 86_400_000);
  const restriction = await tx.driverRestriction.create({
    data: {
      tenantId,
      driverId,
      type: "TEMPORARY",
      startDate,
      endDate,
      reason,
    },
  });

  return { restrictionId: restriction.id, cancelledShiftCount: cancelResult.count };
}
```
