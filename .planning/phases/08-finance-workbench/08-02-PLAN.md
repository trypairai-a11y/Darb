**Create `backend/src/services/finance/reconciliation.ts`:**

```typescript
import { Prisma } from "@prisma/client";
import { prisma } from "../../config";

type Bucket = { threshold: number; count: number; totalGap: string };

export async function buildReconciliation(args: {
  tenantId: string;
  today: Date;
  platform?: "KEETA" | "TALABAT" | "DELIVEROO";
}): Promise<{
  buckets: Bucket[];
  platform: "KEETA" | "TALABAT" | "DELIVEROO" | "ALL";
  total: string;
}> {
  const { tenantId, today, platform } = args;
  const thresholds = [0, 3, 7, 14] as const;

  // Build where clause — tenant FIRST, then status, then platform filter.
  // AMERICANA is EXPLICITLY excluded at the SQL level (CON-cash-platform-coverage):
  //   - when ?platform is set, scope to that platform
  //   - when no filter, exclude AMERICANA via driver relation filter
  const where: Prisma.CashRecordWhereInput = {
    tenantId,
    status: { in: ["PENDING", "PARTIAL"] },
    ...(platform
      ? { driver: { platform } }
      : { driver: { platform: { not: "AMERICANA" } } }),
  };

  const rows = await prisma.cashRecord.findMany({
    where,
    select: {
      date: true,
      salesAmount: true,
      collectionAmount: true,
      driver: { select: { id: true, platform: true } },
    },
    orderBy: { date: "asc" },
  });

  // Bucket rows by age. Use Prisma.Decimal arithmetic to preserve KD precision.
  const buckets: Bucket[] = thresholds.map((threshold, i) => {
    const next = (thresholds[i + 1] ?? Number.POSITIVE_INFINITY) as number;
    const filtered = rows.filter((r) => {
      const ageDays = Math.floor(
        (today.getTime() - r.date.getTime()) / 86400_000,
      );
      return ageDays >= threshold && ageDays < next;
    });
    const totalGapDec = filtered.reduce(
      (acc, r) =>
        acc.plus(new Prisma.Decimal(r.salesAmount).minus(r.collectionAmount)),
      new Prisma.Decimal(0),
    );
    return {
      threshold,
      count: filtered.length,
      totalGap: totalGapDec.toFixed(3),
    };
  });

  const totalDec = buckets.reduce(
    (acc, b) => acc.plus(new Prisma.Decimal(b.totalGap)),
    new Prisma.Decimal(0),
  );

  return {
    buckets,
    platform: platform ?? "ALL",
    total: totalDec.toFixed(3),
  };
}
```

export function ReconciliationGrid({ initial }: { initial: ReconciliationResponse }) {

    refetchInterval: 60_000,  // 1 minute auto-refresh during the morning ritual
  });

  if (isLoading && !data) return <GridSkeleton />;
  if (error) return <div className="text-rose-600">Failed to load reconciliation.</div>;

    <div>
      <PlatformFilterPills value={platform} onChange={setPlatform} />

      </div>
    </div>
  );
}
