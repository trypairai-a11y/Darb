```typescript
import { Prisma } from "@prisma/client";
import { prisma } from "../../config";

export async function computeWeeklyPay(args: {
  tenantId: string;
  driverId: string;
  periodStart: Date;
  periodEnd: Date;
}): Promise<{
  base: string;
  incentives: string;
  deductions: string;
  net: string;
}> {
  const { tenantId, driverId, periodStart, periodEnd } = args;

  const driver = await prisma.driver.findFirst({
    where: { id: driverId, tenantId },
    select: { id: true, contractedRateKd: true },  // verify column name in schema.prisma
  });
  if (!driver) throw new Error("Driver not in tenant scope");

  // Base = contracted rate * total actual hours
  const shifts = await prisma.shift.findMany({
    where: {
      tenantId,
      driverId,
      date: { gte: periodStart, lt: periodEnd },
      actualHoursMinutes: { not: null },
    },
    select: { actualHoursMinutes: true },
  });
  const totalMinutes = shifts.reduce((s, sh) => s + (sh.actualHoursMinutes ?? 0), 0);
  const totalHours = new Prisma.Decimal(totalMinutes).div(60);
  const rate = new Prisma.Decimal(driver.contractedRateKd ?? 0);
  const base = rate.mul(totalHours);

  // W7 resolution: incentives column omitted from PayrollLine. Phase 11 founder-config
  // will add Tenant.payrollIncentivesKd or re-introduce per-line incentives. For Wave 3,
  // payrollComputer returns only {base, deductions, net} — no incentives field.

  // Deductions = sum of EFFECTIVE Penalty.penaltyValue in window
  const penalties = await prisma.penalty.findMany({
    where: {
      tenantId,
      driverId,
      penaltyStatus: "EFFECTIVE",
      createdAt: { gte: periodStart, lt: periodEnd },
    },
    select: { penaltyValue: true },
  });
  const deductions = penalties.reduce(
    (acc, p) => acc.plus(new Prisma.Decimal(p.penaltyValue ?? 0)),
    new Prisma.Decimal(0),
  );

  // Net = base - deductions (clamped >= 0; W7 resolution — no incentives in Wave 3)
  let net = base.minus(deductions);
  if (net.lessThan(0)) net = new Prisma.Decimal(0);

  return {
    base: base.toFixed(3),
    deductions: deductions.toFixed(3),
    net: net.toFixed(3),
  };
}

export async function buildPayrollRun(
  tenantId: string,
  period: string,         // ISO week e.g. "2026-W17"
  userId: string,
): Promise<{ runId: string; lineCount: number; totalNetKd: string }> {
  const { periodStart, periodEnd } = isoWeekToRange(period);

  return prisma.$transaction(async (tx) => {
    const run = await tx.payrollRun.create({
      data: {
        tenantId,
        period,
        periodStart,
        periodEnd,
        status: "DRAFT",
        totalNetKd: 0,
      },
    });

    const drivers = await tx.driver.findMany({
      where: { tenantId, status: "ACTIVE" },
      select: { id: true, iban: true },  // verify column name
    });

    let totalNet = new Prisma.Decimal(0);
    let lineCount = 0;
    for (const d of drivers) {
      const pay = await computeWeeklyPay({ tenantId, driverId: d.id, periodStart, periodEnd });
      if (new Prisma.Decimal(pay.net).equals(0)) continue;  // skip zero-pay drivers
      await tx.payrollLine.create({
        data: {
          tenantId,
          runId: run.id,
          driverId: d.id,
          baseKd: pay.base,
          deductionsKd: pay.deductions,
          netKd: pay.net,
          iban: d.iban ?? null,  // Wave 1 added Driver.iban — nullable; bank file falls back to empty string for missing IBANs
        },
      });
      totalNet = totalNet.plus(pay.net);
      lineCount += 1;
    }

    await tx.payrollRun.update({
      where: { id: run.id },
      data: { totalNetKd: totalNet.toFixed(3) },
    });

    return { runId: run.id, lineCount, totalNetKd: totalNet.toFixed(3) };
  });
}

function isoWeekToRange(period: string): { periodStart: Date; periodEnd: Date } {
  // Parse "YYYY-Www" into the Monday-Sunday range
  const match = /^(\d{4})-W(\d{1,2})$/.exec(period);
  if (!match) throw new Error(`Invalid ISO week period: ${period}`);
  const year = Number(match[1]);
  const week = Number(match[2]);
  // ISO 8601: Week 1 contains the year's first Thursday
  const jan4 = new Date(Date.UTC(year, 0, 4));
  const dow = jan4.getUTCDay() || 7;
  const week1Mon = new Date(jan4);
  week1Mon.setUTCDate(jan4.getUTCDate() - dow + 1);
  const periodStart = new Date(week1Mon);
  periodStart.setUTCDate(week1Mon.getUTCDate() + (week - 1) * 7);
  const periodEnd = new Date(periodStart);
  periodEnd.setUTCDate(periodStart.getUTCDate() + 7);
  return { periodStart, periodEnd };
}
```

```typescript
import type { Writable } from "node:stream";
import Papa from "papaparse";
import { Prisma } from "@prisma/client";
import { prisma } from "../../config";

export async function streamPayrollCsv(
  runId: string,
  tenantId: string,
  out: Writable,
): Promise<{ rowCount: number; totalKd: string }> {
  // Tenant-scope check on the run itself
  const run = await prisma.payrollRun.findFirst({
    where: { id: runId, tenantId },
    select: { id: true, period: true },
  });
  if (!run) throw new Error("Run not in tenant scope");

  // Header
  out.write("IBAN,Amount,Reference\n");

  let rowCount = 0;
  let totalKd = new Prisma.Decimal(0);

  // Stream lines in cursor batches (Prisma's cursor-based iteration)
  const PAGE = 200;
  let cursor: string | undefined;
  for (;;) {
    const batch = await prisma.payrollLine.findMany({
      where: { tenantId, runId },
      orderBy: { id: "asc" },
      take: PAGE,
      ...(cursor ? { skip: 1, cursor: { id: cursor } } : {}),
      select: { id: true, iban: true, netKd: true },
    });
    if (batch.length === 0) break;

    for (const line of batch) {
      if (new Prisma.Decimal(line.netKd).equals(0)) continue;
      const csv = Papa.unparse(
        [{ iban: line.iban ?? "", amount: new Prisma.Decimal(line.netKd).toFixed(3), reference: `RUN-${run.period}` }],
        { header: false },
      );
      out.write(csv + "\n");
      rowCount += 1;
      totalKd = totalKd.plus(line.netKd);
    }
    cursor = batch[batch.length - 1]!.id;
    if (batch.length < PAGE) break;
  }

  out.end();
  return { rowCount, totalKd: totalKd.toFixed(3) };
}
```

**Step 4 — Create `backend/src/services/finance/pnlAggregator.ts`:**

```typescript
import { Prisma } from "@prisma/client";
import { prisma } from "../../config";

const PLATFORMS = ["KEETA", "TALABAT", "DELIVEROO", "AMERICANA"] as const;
const CATEGORIES = ["FUEL", "MAINTENANCE", "RENT", "OFFICE_SALARY", "UTILITIES", "OTHER"] as const;

export async function aggregatePnl(args: {
  tenantId: string;
  period: "month" | "quarter";
  from: Date;
}): Promise<{
  revenue: { byPlatform: Record<typeof PLATFORMS[number], string> };
  expenses: { byCategory: Record<typeof CATEGORIES[number], string> };
  net: string;
}> {
  const { tenantId, period, from } = args;
  const months = period === "month" ? 1 : 3;
  const to = new Date(from);
  to.setUTCMonth(from.getUTCMonth() + months);

  // W3 resolution: Revenue = sum of Invoice.totalKd by platform WHERE status='ISSUED'.
  // Tenant filter REQUIRED. Phase 8 does NOT auto-ingest invoices from delivery platforms
  // (that is Phase 10), so this returns 0.000 across all platforms until the founder manually
  // creates Invoice rows via POST /api/finance/invoices and transitions them to ISSUED.
  const revenueRows = await prisma.invoice.groupBy({
    by: ["platform"],
    where: { tenantId, status: "ISSUED", periodStart: { gte: from, lt: to } },
    _sum: { totalKd: true },
  });
  const revenueByPlatform = Object.fromEntries(
    PLATFORMS.map((p) => [p, "0.000"]),
  ) as Record<typeof PLATFORMS[number], string>;
  for (const row of revenueRows) {
    revenueByPlatform[row.platform as typeof PLATFORMS[number]] =
      new Prisma.Decimal(row._sum.totalKd ?? 0).toFixed(3);
  }

  // Expenses = sum of Expense.amountKd by category in window; exclude soft-deleted
  const expenseRows = await prisma.expense.groupBy({
    by: ["category"],
    where: { tenantId, deletedAt: null, dateIncurred: { gte: from, lt: to } },
    _sum: { amountKd: true },
  });
  const expensesByCategory = Object.fromEntries(
    CATEGORIES.map((c) => [c, "0.000"]),
  ) as Record<typeof CATEGORIES[number], string>;
  for (const row of expenseRows) {
    expensesByCategory[row.category as typeof CATEGORIES[number]] =
      new Prisma.Decimal(row._sum.amountKd ?? 0).toFixed(3);
  }

  // Net = sum(revenue) - sum(expenses)
  const totalRevenue = Object.values(revenueByPlatform).reduce(
    (acc, v) => acc.plus(v),
    new Prisma.Decimal(0),
  );
  const totalExpenses = Object.values(expensesByCategory).reduce(
    (acc, v) => acc.plus(v),
    new Prisma.Decimal(0),
  );
  const net = totalRevenue.minus(totalExpenses);

  return {
    revenue: { byPlatform: revenueByPlatform },
    expenses: { byCategory: expensesByCategory },
    net: net.toFixed(3),
  };
}
```
