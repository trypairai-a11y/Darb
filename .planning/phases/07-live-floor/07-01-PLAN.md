import { z } from "zod";
import { prisma } from "../../../config";
import { defineTool, toolRegistry } from "../../registry";

/**
 * Phase 7 read tool — REQ-floor-live-map.
 *
 * Returns drivers with today's KeetaDailyMetrics.rejectedAuto >= 3, the
 * canonical "order-rejection ×3+" pill signal. Tenant-scoped via existing
 * prismaExtensions.ts::hasTenantFilter. Keeta-only in v1 (W3: documented
 * limitation; the cross-platform OrderEvent.action='REJECT' aggregate is

 *
 * Why a separate tool from liveFleetStatus: the pill click-through opens
 * a list overlay (resolution #2: both behaviors); the chat agent also
 * benefits from "show me rejection-heavy drivers today" as a typed query.
 */
export const orderRejectionToday = defineTool({
  name: "orderRejectionToday",
  description:
    "Return drivers with today's order-rejection count >= 3 (Keeta auto-rejections only in v1). Each item is {driverId, name, count}. Ordered by count DESC. Tenant-scoped. Use this for the Floor pill click-through list view and for chat queries about rejection-heavy drivers.",
  inputSchema: {
    type: "object" as const,
    properties: {},
    required: [],
    additionalProperties: false,
  },
  inputValidator: z.object({}).strict(),
  strict: true,
  sideEffect: "read",
  requiredRole: ["ADMIN", "OPS_MANAGER", "SUPERVISOR", "ACCOUNTANT", "VIEWER"],
  requiresApproval: false,
  allowedAgents: ["*"],
  async execute(ctx) {
    const todayStart = new Date();
    todayStart.setHours(0, 0, 0, 0);
    const todayEnd = new Date(todayStart.getTime() + 86_400_000);
    try {
      const rows = await prisma.keetaDailyMetrics.findMany({
        where: {
          tenantId: ctx.tenantId,
          date: { gte: todayStart, lt: todayEnd },
          rejectedAuto: { gte: 3 },
        },
        select: {
          driverId: true,
          rejectedAuto: true,
          driver: { select: { name: true } },
        },
        orderBy: { rejectedAuto: "desc" },
      });
      return (rows ?? []).map((r) => ({
        driverId: r.driverId,
        name: r.driver?.name ?? "(unknown)",
        count: r.rejectedAuto,
      }));
    } catch {
      return [];
    }
  },
});
