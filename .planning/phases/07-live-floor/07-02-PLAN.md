```typescript
// Phase 7 — Single-source TypeScript shapes for the Live Floor surface.
// Both Wave 2 (map + marker + hook) and Wave 3 (page + filters + panel + ping button)
// import from here. Backend contracts come from backend/src/routes/floor.ts.

export type Platform = "KEETA" | "TALABAT" | "DELIVEROO" | "AMERICANA";

export interface CourierTodayStats {
  ordersDelivered: number;
  ordersRejected: number;
  onlineMinutes: number;
}

/**
 * Matches GET /api/floor/snapshot response item (backend Wave 1).
 * Wave 3 adds optional `scheduledShiftStart` for the blue "scheduled-not-online"
 * status in the panel; for the initial snapshot it's always missing/absent.
 */
export interface CourierSnapshot {
  driverId: string;
  name: string;
  phone: string | null;
  platform: Platform | null;
  vehicleType: "MOTORCYCLE" | "CAR" | null;
  area: string | null;
  lat: number | null;       // null = no GPS yet
  lng: number | null;
  lastGpsAt: string | null; // ISO timestamp
  isOnline: boolean;
  todayStats: CourierTodayStats;
  scheduledShiftStart?: string | null;
}

export interface FloorCounters {
  scheduledNotOnline: number;
  gpsStale: number;
  orderRejection: number;
}

export type CourierStatus = "WORKING" | "IDLE" | "STALE" | "SCHEDULED_NOT_ONLINE";

/**
 * Per CON-floor-counters: stale = lastGpsAt > 10min ago.
 * Per CON-floor-dot-colors: green=working (online+fresh+orders>0),
 * grey=idle (online+fresh+orders=0), red=stale, blue=scheduled-not-online.
 */
export function classifyCourier(
  courier: CourierSnapshot,
  nowMs: number = Date.now(),
): CourierStatus {
  if (!courier.isOnline) return "SCHEDULED_NOT_ONLINE";
  if (!courier.lastGpsAt) return "STALE"; // no GPS ever — treat as stale
  const ageMs = nowMs - new Date(courier.lastGpsAt).getTime();
  if (ageMs > 10 * 60 * 1000) return "STALE";
  return courier.todayStats.ordersDelivered > 0 ? "WORKING" : "IDLE";
}

export interface FloorFilter {
  status?: CourierStatus;
  platforms?: Platform[];
  area?: string;
  q?: string;
}

/**
 * Mirrors backend Wave 1's pingBodySchema (routes/floor.ts).
 * Phase 7 ships only these three; broader intents from draftCourierMessage stay
 * available via the agent's chat path.
 */
export type PingIntent = "WARN_GPS_STALE" | "WARN_ORDER_REJECTIONS" | "WARN_LATE_CLOCKIN";

/** SSE event payload shapes (Wave 1's publishEvent contracts). */

  type: "gps_point";
  tenantId: string;
  payload: {
    driverId: string;
    lat: number;
    lng: number;
    capturedAt: string;
    batteryPct: number | null;
  };
  timestamp: string;
}

  type: "online_session_update";
  tenantId: string;
  payload: {
    driverId: string;
    isOnline: boolean;
    area: string | null;
  };
  timestamp: string;
}

export type FloorSseEvent = GpsPointEvent | OnlineSessionUpdateEvent;
```

```typescript
// Phase 7 Wave 2 — courier dot on the Live Floor map.
// Uses L.divIcon (Pattern 7) so the marker is HTML, themeable via Tailwind tokens,
// and bubbles click events to onSelect.
//
// B3: platform ring class is read from the STATIC PLATFORM_RING map in
// types/floor.ts — no template-literal class construction. Tailwind's JIT
// scans source for literal class names; runtime-built strings are stripped.
"use client";
import { Marker, Popup } from "react-leaflet";
import L from "leaflet";
import { useMemo } from "react";
import type { CourierSnapshot, CourierStatus } from "@/types/floor";
import { classifyCourier, PLATFORM_RING } from "@/types/floor";

const STATUS_BG: Record<CourierStatus, string> = {
  WORKING: "bg-emerald-500",
  IDLE: "bg-sand-400",
  STALE: "bg-red-500",
  SCHEDULED_NOT_ONLINE: "bg-blue-500",
};

export interface CourierMarkerProps {
  courier: CourierSnapshot;
  onSelect: (driverId: string) => void;
  /** Override the "now" reference for deterministic testing. */
  nowMs?: number;
}

/**
 * Marker for one courier. Renders only if lat+lng are available; otherwise
 * caller should filter the marker out (a courier with no GPS yet is shown
 * in the right-rail list view but cannot be placed on the map).
 */
export function CourierMarker({ courier, onSelect, nowMs }: CourierMarkerProps) {
  const status = classifyCourier(courier, nowMs ?? Date.now());
  const platform = courier.platform;

  const icon = useMemo(() => {
    const statusClass = STATUS_BG[status];
    // B3: static map lookup — Tailwind sees ring-keeta/ring-talabat/etc. as
    // plain source tokens in types/floor.ts and emits the CSS rules.
    const ringClass = platform ? PLATFORM_RING[platform] : "ring-sand-300";
    return L.divIcon({
      html: `<div class="${statusClass} ${ringClass} ring-2 ring-offset-1 ring-offset-white rounded-full w-3.5 h-3.5" aria-label="${courier.name}"></div>`,
      className: "",  // suppress default Leaflet styles
      iconSize: [14, 14],
      iconAnchor: [7, 7],
    });
  }, [status, platform, courier.name]);

  if (courier.lat == null || courier.lng == null) return null;

  return (
    <Marker
      position={[courier.lat, courier.lng]}
      icon={icon}
      eventHandlers={{ click: () => onSelect(courier.driverId) }}
    >
      <Popup>
        <div className="text-xs">
          <div className="font-semibold">{courier.name}</div>
          {courier.area && <div className="text-sand-600">{courier.area}</div>}
          <div className="text-sand-500">{status.replace("_", " ").toLowerCase()}</div>
        </div>
      </Popup>
    </Marker>
  );
}

export default CourierMarker;
```

```typescript
}
```

**Step 3 — `frontend/src/hooks/useFloorRealtime.ts` (~95 lines — NIT2 cleanup):**

```typescript
// Phase 7 Wave 2 — composes useSSE + react-query to drive <LiveFloorMap/>.
// Couriers are held in a ref Map for surgical position updates on gps_point;
// counters are polled every 30s and invalidated on online_session_update.
// visibilitychange triggers a snapshot refetch (Pitfall 10).
"use client";
import { useEffect, useRef, useState, useCallback } from "react";
import { useQuery, useQueryClient } from "@tanstack/react-query";
import api from "@/lib/api";
import { useSSE } from "@/hooks/useSSE";
import type {
  CourierSnapshot,
  FloorCounters,
  GpsPointEvent,
  FloorSseEvent,
} from "@/types/floor";

export interface UseFloorRealtimeReturn {
  couriers: CourierSnapshot[];
  counters: FloorCounters | undefined;
  connected: boolean;
  refetchSnapshot: () => Promise<unknown>;
}

export function useFloorRealtime(): UseFloorRealtimeReturn {
  const queryClient = useQueryClient();

  // Initial snapshot — react-query holds the canonical list.
  const snapshotQuery = useQuery<{ couriers: CourierSnapshot[] }>({
    queryKey: ["floorSnapshot"],
    queryFn: async () => {
      const res = await api.get("/api/floor/snapshot");
      return res.data;
    },
    staleTime: 60_000,
  });

  // Counters — polled every 30s with a 25s staleTime to avoid double-fetch.
  const countersQuery = useQuery<FloorCounters>({
    queryKey: ["floorCounters"],
    queryFn: async () => {
      const res = await api.get("/api/floor/counters");
      return res.data;
    },
    refetchInterval: 30_000,
    staleTime: 25_000,
  });

  // Surgical position updates: a ref-held Map keyed by driverId avoids
  // re-rendering all ~300 markers per gps_point event (Anti-Pattern guard).
  // We bump a counter to trigger a single render when the dict changes.
  const positionsRef = useRef<Map<string, { lat: number; lng: number; lastGpsAt: string }>>(new Map());
  const [renderTick, setRenderTick] = useState(0);

  const onSseMessage = useCallback(
    (data: unknown) => {
      const evt = data as FloorSseEvent | { type: string };
      if (!evt || typeof evt !== "object" || !("type" in evt)) return;
      if (evt.type === "gps_point") {
        const e = evt as GpsPointEvent;
        positionsRef.current.set(e.payload.driverId, {
          lat: e.payload.lat,
          lng: e.payload.lng,
          lastGpsAt: e.payload.capturedAt,
        });
        setRenderTick((t) => t + 1);
      } else if (evt.type === "online_session_update") {
        queryClient.invalidateQueries({ queryKey: ["floorCounters"] });
        // Also nudge the snapshot — a new driver may be coming online.
        queryClient.invalidateQueries({ queryKey: ["floorSnapshot"] });
      }
    },
    [queryClient],
  );

  const sse = useSSE({ url: "/api/events", onMessage: onSseMessage });

  // Pitfall 10 — refetch on visibility regain.
  useEffect(() => {
    const onVisibility = () => {
      if (document.visibilityState === "visible") {
        queryClient.invalidateQueries({ queryKey: ["floorSnapshot"] });
        queryClient.invalidateQueries({ queryKey: ["floorCounters"] });
      }
    };
    document.addEventListener("visibilitychange", onVisibility);
    return () => document.removeEventListener("visibilitychange", onVisibility);
  }, [queryClient]);

  // Merge ref-held position overrides onto the snapshot. Stable iteration
  // order is preserved from the snapshot; markers that aren't in the dict
  // pass through unchanged.
  const couriers = (snapshotQuery.data?.couriers ?? []).map((c) => {
    const override = positionsRef.current.get(c.driverId);
    if (!override) return c;
    return { ...c, lat: override.lat, lng: override.lng, lastGpsAt: override.lastGpsAt };
  });

  // NIT2: acknowledge renderTick so React keeps the hook output in sync with
  // ref mutations — the variable IS the dependency signal. Plain `void`
  // statement; no spread of an empty record onto the return value.
  void renderTick;

  return {
    couriers,
    counters: countersQuery.data,
    connected: sse.connected,
    refetchSnapshot: () => snapshotQuery.refetch(),
  };
}
```
